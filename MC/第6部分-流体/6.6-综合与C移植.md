## 6.6.2 C++ 工程蓝图

```cpp
// === 第 6 部分：流体动力学 ===

namespace mc::fluid {

// 6.1 Fluid 抽象
class Fluid {
public:
    Fluid();
    virtual ~Fluid() = default;
    
    const FluidState& defaultFluidState() const { return *defaultState_; }
    
    virtual Item* getBucket() const = 0;
    virtual void tick(ServerLevel& level, BlockPos pos, const BlockState& blockState, const FluidState& fluidState) {}
    virtual void randomTick(ServerLevel& level, BlockPos pos, const FluidState& state, RandomSource& random) {}
    virtual Vec3 getFlow(BlockGetter& level, BlockPos pos, const FluidState& state) const = 0;
    virtual int getTickDelay(const LevelReader& level) const = 0;
    virtual const BlockState* createLegacyBlock(const FluidState& state) const = 0;
    virtual bool isEmpty() const { return false; }
    virtual bool isRandomlyTicking() const { return false; }
    virtual bool isSource(const FluidState& state) const = 0;
    virtual bool isSame(Fluid* other) const { return this == other; }
    
protected:
    std::unique_ptr<StateDefinition<Fluid, FluidState>> states_;
    const FluidState* defaultState_;
};

class FluidState {
public:
    Fluid* getType() const { return owner_; }
    bool isSource() const { return owner_->isSource(*this); }
    bool isEmpty() const { return owner_->isEmpty(); }
    int getAmount() const;
    float getOwnHeight() const;
    Vec3 getFlow(BlockGetter& level, BlockPos pos) const { return owner_->getFlow(level, pos, *this); }
    bool is(const TagKey<Fluid>& tag) const;
    bool is(Fluid* fluid) const { return owner_ == fluid; }
    
    template<typename T> T getValue(const Property* prop) const;
    const FluidState* setValue(const Property* prop, ...) const;
    
private:
    Fluid* owner_;
    boost::container::flat_map<const Property*, std::any> values_;
    int32_t globalId_;
};

class EmptyFluid : public Fluid {
public:
    bool isEmpty() const override { return true; }
    bool isSource(const FluidState&) const override { return false; }
    Vec3 getFlow(BlockGetter&, BlockPos, const FluidState&) const override { return Vec3::ZERO; }
    int getTickDelay(const LevelReader&) const override { return 0; }
    // ⋯
};

// 6.2 FlowingFluid
class FlowingFluid : public Fluid {
public:
    static const BooleanProperty FALLING;
    static const IntegerProperty LEVEL;     // 1..8
    
    virtual Fluid* getFlowing() = 0;
    virtual Fluid* getSource() = 0;
    
    const FluidState& getFlowing(int level, bool falling);
    const FluidState& getSource(bool falling);
    
    virtual bool canConvertToSource(ServerLevel& level) = 0;
    virtual void beforeDestroyingBlock(LevelAccessor& level, BlockPos pos, const BlockState& state) = 0;
    virtual int getSlopeFindDistance(const LevelReader& level) = 0;
    virtual int getDropOff(const LevelReader& level) = 0;
    
    void tick(ServerLevel& level, BlockPos pos, const BlockState& blockState, const FluidState& fluidState) override;
    Vec3 getFlow(BlockGetter& level, BlockPos pos, const FluidState& state) const override;

protected:
    void spread(ServerLevel& level, BlockPos pos, const BlockState& blockState, const FluidState& fluidState);
    void spreadTo(LevelAccessor& level, BlockPos pos, const BlockState& state, Direction dir, const FluidState& fluidState);
    
    // 6.3 距离场扩散算法
    std::map<Direction, FluidState> getSpread(ServerLevel& level, BlockPos pos, const BlockState& state);
    int getSlopeDistance(ServerLevel& level, BlockPos pos, int currentDistance, Direction skipDir, ...);
    bool isWaterHole(...) const;
    
    FluidState getNewLiquid(ServerLevel& level, BlockPos pos, const BlockState& state);
    int sourceNeighborCount(LevelReader& level, BlockPos pos);
};

// 6.4 WaterFluid
class WaterFluid : public FlowingFluid {
public:
    int getSlopeFindDistance(const LevelReader&) override { return 4; }
    int getDropOff(const LevelReader&) override { return 1; }
    int getTickDelay(const LevelReader&) override { return 5; }
    bool canConvertToSource(ServerLevel& level) override { 
        return level.getGameRules().getBoolean(GameRules::WATER_SOURCE_CONVERSION);
    }
    void beforeDestroyingBlock(LevelAccessor& level, BlockPos pos, const BlockState& state) override {
        Block::dropResources(state, level, pos, ...);
    }
    
    class Source : public WaterFluid {
    public:
        bool isSource(const FluidState&) const override { return true; }
    };
    
    class Flowing : public WaterFluid {
    public:
        bool isSource(const FluidState&) const override { return false; }
        // 创建时多 add LEVEL 属性
    };
};

// 6.4 LavaFluid
class LavaFluid : public FlowingFluid {
public:
    bool isRandomlyTicking() const override { return true; }
    
    int getSlopeFindDistance(const LevelReader& level) override {
        return level.dimensionType().ultraWarm() ? 4 : 2;
    }
    int getDropOff(const LevelReader& level) override {
        return level.dimensionType().ultraWarm() ? 1 : 2;
    }
    int getTickDelay(const LevelReader& level) override {
        return level.dimensionType().ultraWarm() ? 10 : 30;
    }
    
    void randomTick(ServerLevel& level, BlockPos pos, const FluidState& state, RandomSource& random) override {
        // 引发火灾 / 自燃
    }
};

// 6.1 Fluids 注册表
namespace Fluids {
    extern Fluid* EMPTY;
    extern FlowingFluid* WATER;
    extern FlowingFluid* FLOWING_WATER;
    extern FlowingFluid* LAVA;
    extern FlowingFluid* FLOWING_LAVA;
}

}

namespace mc::block {

// 6.1 LiquidBlock
class LiquidBlock : public Block, public BucketPickup {
public:
    static const IntegerProperty LEVEL;
    
    LiquidBlock(FlowingFluid* fluid, BlockBehaviour::Properties props);
    
    const FluidState& getFluidState(const BlockState& state) const override {
        int blockLevel = state.getValue(LEVEL);
        if (blockLevel >= 8) return fluid_->getFlowing(8, true);
        if (blockLevel == 0) return fluid_->getSource(false);
        return fluid_->getFlowing(8 - blockLevel, false);
    }
    
    void neighborChanged(...) override {
        if (shouldSpreadLiquid(level, pos, state)) {
            level.scheduleTick(pos, state.getFluidState().getType(), fluid_->getTickDelay(level));
        }
    }
    
    bool isPathfindable(const BlockState& state, PathComputationType type) const override {
        return !fluid_->is(FluidTags::LAVA);
    }
    
private:
    FlowingFluid* fluid_;
};

// 6.5 SimpleWaterloggedBlock
class SimpleWaterloggedBlock : public LiquidBlockContainer {
public:
    static const BooleanProperty WATERLOGGED;
    
    bool canPlaceLiquid(...) override {
        return !state.getValue(WATERLOGGED) && fluid == Fluids::WATER;
    }
    
    bool placeLiquid(LevelAccessor& level, BlockPos pos, const BlockState& state, const FluidState& fluidState) override {
        if (!state.getValue(WATERLOGGED) && fluidState.getType() == Fluids::WATER) {
            if (!level.isClientSide()) {
                level.setBlock(pos, state.setValue(WATERLOGGED, true), 3);
                level.scheduleTick(pos, fluidState.getType(), fluidState.getType()->getTickDelay(level));
            }
            return true;
        }
        return false;
    }
};

// 6.5 SpongeBlock
class SpongeBlock : public Block {
public:
    void onPlace(...) override { tryAbsorbWater(level, pos); }
    void neighborChanged(...) override { tryAbsorbWater(level, pos); }
    
private:
    bool removeWaterBreadthFirstSearch(Level& level, BlockPos pos);
    void tryAbsorbWater(Level& level, BlockPos pos);
};

}
```

## 6.6.4 vanilla 嵌入式 vs SoA 优化

vanilla 的设计**FluidState 嵌入 BlockState**——同一格只有一个 BlockState 引用。这有性能问题：

- **优点**：内存紧凑（不为流体增加存储）。
- **缺点**：每次"是不是水"查询要走 BlockState.getFluidState（虚函数）。

**性能优化**：用**单独流体层**：

```cpp
class LevelChunkSection {
    PalettedContainer<const BlockState*> states;
    PalettedContainer<const BlockState*> blocks;     // 不变
    PalettedContainer<const FluidState*> fluids;     // ★ 单独
};
```

**两个独立 PalettedContainer**——一格能同时是 fence + water。

**好处**：

- isWater 查询从虚函数变为指针比较：5x 加速。
- 流体 tick 遍历 SoA 友好（只看 fluids container）。
- waterlogged 不需要属性 + 多个 BlockState。

**代价**：

- 内存翻倍（每 section 多 ~2 KB）。
- 协议复杂——必须额外发流体 palette。
- 持久化复杂——SerializableChunkData 多一字段。

**Mojang 的选择**：嵌入式 —— 协议兼容性优先。modder 想优化可以 mixin。

**C++ 引擎选择**：第一版照搬嵌入式。性能瓶颈再考虑分离。

## 6.6.5 ECS vs OOP

流体方块**少（仅水 / 熔岩）**且**行为差异较大**——OOP 适合：

- WaterFluid / LavaFluid 子类化。
- waterloggable 用接口扩展现有方块。

**水量 / 流向 vector 适合 SoA**：

- chunk section 的 fluid LEVEL 数组连续访问。
- getFlow 计算可批处理。

**实际**：第一版 OOP 即可。瓶颈定位后再 SoA 化。

## 6.6.6 性能 budget

```
单 chunk 单 tick fluid 处理:
- random tick (lava randomTick): ~5 微秒
- scheduled fluid tick: ~50 个/chunk × 30 微秒/tick = 1.5 ms

视距 32 chunk² ≈ 1024 chunk × 1.5 ms / chunk = 1.5 秒
但实际：稳态后大部分流体不变化（source 不 spread）
有效 tick：~10-50 个/秒/chunk

总 fluid tick budget：65536（与 block tick 独立）
正常运行远低于此。
极端（玩家爆破水底）：数千 fluid tick / tick，触发 budget 限制。
```

## 6.6.9 一句话总结

第 6 部分讲清楚 Minecraft 流体作为**第 4 部分方块更新机制的距离场扩散特化**——`Fluid` / `FluidState` 体系完全镜像 `Block` / `BlockState`，5 个 vanilla Fluid（EMPTY / WATER / FLOWING_WATER / LAVA / FLOWING_LAVA）；FluidState 嵌入 BlockState（LiquidBlock 直接持有 / waterloggable 通过 WATERLOGGED 属性）；FlowingFluid 用 LEVEL (1-8) + FALLING 双属性，主世界水扩散 7 格 / 5 tick，主世界熔岩 4 格 / 30 tick，下界熔岩 7 格 / 10 tick；扩散用 BFS 距离场寻找最近"hole"（每方向最多深度 4 / 2）；waterloggable 60+ 方块通过 SimpleWaterloggedBlock 接口；水 + 熔岩 → obsidian / cobblestone / stone；海绵 BFS 吸 64 格水；下界水蒸发；C++ 移植蓝图明确，第一版 OOP 嵌入式即可，性能瓶颈再 SoA 优化。
