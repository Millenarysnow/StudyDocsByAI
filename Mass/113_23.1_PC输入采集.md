## 23.1.1 PC 的"职责墙"

```
ARTSPlayerController:
   ✓ 接收输入(键盘 / 鼠标)
   ✓ 维护本地选中(LocalPlayerSubsystem)
   ✓ 射线检测命中 entity
   ✓ 编组(Ctrl+1)
   ✓ 把命令 RPC 发给 server
   ✓ 接收 server 通知(击杀提示等)
   
   ✗ 不直接修改 Mass entity 状态
   ✗ 不直接修改其它玩家的数据
   ✗ 不维护"游戏全局"状态(那是 GameState)
```

## 23.1.2 完整 PC 实现示例

```cpp
UCLASS()
class ARTSPlayerController : public APlayerController
{
    GENERATED_BODY()
public:
    virtual void SetupInputComponent() override
    {
        Super::SetupInputComponent();
        
        if (UEnhancedInputComponent* EIC = Cast<UEnhancedInputComponent>(InputComponent))
        {
            EIC->BindAction(IA_Select, ETriggerEvent::Started, this, &ThisClass::OnSelectStart);
            EIC->BindAction(IA_Select, ETriggerEvent::Completed, this, &ThisClass::OnSelectEnd);
            EIC->BindAction(IA_Command, ETriggerEvent::Started, this, &ThisClass::OnCommand);
            EIC->BindAction(IA_AddSelect, ETriggerEvent::Started, this, &ThisClass::OnAddSelect);
            
            for (int32 i = 1; i <= 9; ++i)
            {
                EIC->BindAction(IA_GroupSet[i], ETriggerEvent::Started, this, &ThisClass::OnSetGroup, i);
                EIC->BindAction(IA_GroupRecall[i], ETriggerEvent::Started, this, &ThisClass::OnRecallGroup, i);
            }
        }
    }
    
    UFUNCTION(Server, Reliable, WithValidation)
    void ServerIssueCommand(const FRTSCommand& Cmd);
    
    UFUNCTION(Client, Reliable)
    void ClientShowKillFeed(const FString& Killer, const FString& Victim);

private:
    bool bIsBoxSelecting = false;
    FVector2D BoxStartPos;
    
    void OnSelectStart(const FInputActionValue& V);
    void OnSelectEnd(const FInputActionValue& V);
    void OnCommand(const FInputActionValue& V);
    void OnAddSelect(const FInputActionValue& V);
    void OnSetGroup(const FInputActionValue& V, int32 GroupID);
    void OnRecallGroup(const FInputActionValue& V, int32 GroupID);
};
```

## 23.1.3 单击选择

```cpp
void ARTSPlayerController::OnSelectStart(const FInputActionValue& V)
{
    bIsBoxSelecting = true;
    GetMousePosition(BoxStartPos.X, BoxStartPos.Y);
    
    // 标记开始(可能后面变框选)
}

void ARTSPlayerController::OnSelectEnd(const FInputActionValue& V)
{
    if (!bIsBoxSelecting) return;
    
    FVector2D BoxEnd;
    GetMousePosition(BoxEnd.X, BoxEnd.Y);
    
    URTSSelectionSubsystem* Sel = GetLocalPlayer()->GetSubsystem<URTSSelectionSubsystem>();
    
    if (FVector2D::Distance(BoxStartPos, BoxEnd) < 5.0f)
    {
        // 单击(几乎没拖)
        FHitResult Hit;
        GetHitResultUnderCursor(ECC_Visibility, false, Hit);
        if (Hit.bBlockingHit)
        {
            // 检测命中 actor 还是 entity
            if (AActor* HitActor = Hit.GetActor())
            {
                Sel->SelectActor(HitActor);
            }
            else
            {
                URTSSpatialIndexSubsystem* SI = GetWorld()->GetSubsystem<URTSSpatialIndexSubsystem>();
                FMassEntityHandle E = SI->QueryNearest(Hit.Location, 100.0f);
                if (E.IsValid()) Sel->SelectEntity(E);
            }
        }
    }
    else
    {
        // 框选
        FBox2D Rect(FVector2D::Min(BoxStartPos, BoxEnd), FVector2D::Max(BoxStartPos, BoxEnd));
        TArray<FMassEntityHandle> Entities = QueryEntitiesInScreenBox(Rect);
        Sel->SetEntities(Entities);
    }
    
    bIsBoxSelecting = false;
}
```

## 23.1.4 命令派发

```cpp
void ARTSPlayerController::OnCommand(const FInputActionValue& V)
{
    URTSSelectionSubsystem* Sel = GetLocalPlayer()->GetSubsystem<URTSSelectionSubsystem>();
    if (Sel->SelectedEntities.IsEmpty()) return;
    
    FHitResult Hit;
    GetHitResultUnderCursor(ECC_Visibility, false, Hit);
    if (!Hit.bBlockingHit) return;
    
    FRTSCommand Cmd;
    Cmd.Targets = Sel->SelectedEntities;
    Cmd.TargetLocation = Hit.Location;
    Cmd.bQueue = IsShiftPressed();    // Shift = 加入命令队列
    
    // 决定命令类型(根据 Hit 上下文)
    if (AActor* HitActor = Hit.GetActor())
    {
        if (IsEnemy(HitActor)) Cmd.Type = ECmdType::Attack;
        else if (IsResource(HitActor)) Cmd.Type = ECmdType::Gather;
        else Cmd.Type = ECmdType::Move;
    }
    else
    {
        // 命中地面
        Cmd.Type = ECmdType::Move;
    }
    
    // 发给 server
    if (HasAuthority()) {
        ServerIssueCommand_Implementation(Cmd);
    } else {
        ServerIssueCommand(Cmd);
    }
    
    // 本地立即视觉反馈(不预测移动)
    SpawnCommandIndicator(Hit.Location, Cmd.Type);
}

bool ARTSPlayerController::ServerIssueCommand_Validate(const FRTSCommand& Cmd) {
    return Cmd.Targets.Num() <= 1000;     // 防作弊
}

void ARTSPlayerController::ServerIssueCommand_Implementation(const FRTSCommand& Cmd)
{
    // 验证拥有权
    ARTSPlayerState* PS = GetPlayerState<ARTSPlayerState>();
    if (!PS) return;
    
    URTSCommandSubsystem* Sub = GetWorld()->GetSubsystem<URTSCommandSubsystem>();
    Sub->EnqueueCommand(PS->TeamID, Cmd);
}
```

## 23.1.5 编组

```cpp
void ARTSPlayerController::OnSetGroup(const FInputActionValue& V, int32 GroupID)
{
    URTSSelectionSubsystem* Sel = GetLocalPlayer()->GetSubsystem<URTSSelectionSubsystem>();
    Sel->SetGroup(GroupID, Sel->SelectedEntities);
}

void ARTSPlayerController::OnRecallGroup(const FInputActionValue& V, int32 GroupID)
{
    URTSSelectionSubsystem* Sel = GetLocalPlayer()->GetSubsystem<URTSSelectionSubsystem>();
    Sel->RecallGroup(GroupID);
}
```
