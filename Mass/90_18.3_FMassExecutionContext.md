## 18.3.1 Ctx 主要成员

```cpp
struct FMassExecutionContext
{
    // 时间
    float DeltaTimeSeconds;
    float GetDeltaTimeSeconds() const;
    
    // 当前 chunk 的 entity 数
    int32 GetNumEntities() const;
    
    // Fragment 数据访问(chunk 级)
    template<typename T> TArrayView<T> GetMutableFragmentView();
    template<typename T> TConstArrayView<T> GetFragmentView() const;
    template<typename T> bool HasFragment() const;
    
    // SharedFragment / ChunkFragment 访问
    template<typename T> T& GetMutableSharedFragment();
    template<typename T> const T& GetConstSharedFragment() const;
    
    // 单 entity 工具
    FMassEntityHandle GetEntity(int32 IndexInChunk) const;
    
    // 命令缓冲
    FMassCommandBuffer& Defer();
    
    // 当前 archetype handle
    FMassArchetypeHandle GetArchetypeHandle() const;
    
    // ...
};
```

## 18.3.2 ForEachEntityChunk vs ForEachEntity

```cpp
// ★ 99% 用 ForEachEntityChunk(快)
Query.ForEachEntityChunk(EM, Ctx, [](FMassExecutionContext& InCtx)
{
    // 拿整 chunk 数组,顺序处理
    auto P = InCtx.GetMutableFragmentView<FPosFragment>();
    for (int32 i = 0; i < InCtx.GetNumEntities(); ++i) P[i].X += DT;
});

// 极少用 ForEachEntity(慢,per-entity 调用回调)
Query.ForEachEntity(EM, Ctx, [](FMassExecutionContext& InCtx, FMassEntityHandle E)
{
    // 每 entity 一次调用,vtable + cache miss
});
```

## 18.3.3 完整 Processor 示例

```cpp
void URTSCombatProcessor::Execute(FMassEntityManager& EM, FMassExecutionContext& Ctx)
{
    Query.ForEachEntityChunk(EM, Ctx, [this](FMassExecutionContext& InCtx)
    {
        const TConstArrayView<FMassPositionFragment> Pos = 
            InCtx.GetFragmentView<FMassPositionFragment>();
        TArrayView<FUnitHealthFragment> Health = 
            InCtx.GetMutableFragmentView<FUnitHealthFragment>();
        
        const FUnitTypeStats& Stats = InCtx.GetConstSharedFragment<FUnitTypeStats>();
        
        const int32 N = InCtx.GetNumEntities();
        const float DT = InCtx.GetDeltaTimeSeconds();
        
        for (int32 i = 0; i < N; ++i)
        {
            // 战斗逻辑
            if (Health[i].HP <= 0)
            {
                FMassEntityHandle E = InCtx.GetEntity(i);
                InCtx.Defer().AddTag<FDeadTag>(E);   // 延迟添加 Tag(下帧迁移 archetype)
                InCtx.Defer().DestroyEntity(E);       // 或销毁
            }
        }
    });
}
```

InCtx.Defer().DestroyEntity(E);
