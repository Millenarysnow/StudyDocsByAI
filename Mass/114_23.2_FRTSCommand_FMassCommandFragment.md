## 23.2.1 FRTSCommand(网络可序列化)

```cpp
USTRUCT()
struct FRTSCommand
{
    GENERATED_BODY()
    
    UPROPERTY()
    ECmdType Type;          // Move / Attack / Gather / Patrol / Stop / Hold
    
    UPROPERTY()
    TArray<FMassEntityHandle> Targets;     // 受影响的 entity
    
    UPROPERTY()
    FVector TargetLocation;
    
    UPROPERTY()
    FMassEntityHandle TargetEntity;        // 对目标 entity(若有,如 Attack)
    
    UPROPERTY()
    int32 PlayerID;                         // 发起玩家
    
    UPROPERTY()
    bool bQueue;                            // 是否加入命令队列
    
    // 网络序列化
    bool NetSerialize(FArchive& Ar, ...);
};

template<>
struct TStructOpsTypeTraits<FRTSCommand> : TStructOpsTypeTraitsBase2<FRTSCommand>
{
    enum { WithNetSerializer = true };
};

bool FRTSCommand::NetSerialize(FArchive& Ar, UPackageMap* Map, bool& bOutSuccess)
{
    Ar.SerializeBits(&Type, 4);    // 4 位足够 16 种命令类型
    
    // 目标 entity 列表(用 NetID 量化)
    int32 N = Targets.Num();
    Ar.SerializeIntPacked(*(uint32*)&N);   // 变长 int
    if (Ar.IsLoading()) Targets.SetNum(N);
    for (auto& T : Targets) {
        // 把 EntityHandle 转 NetID 再传
    }
    
    // 位置量化
    int16 X = (int16)(TargetLocation.X * 100);
    int16 Y = (int16)(TargetLocation.Y * 100);
    int8 Z = (int8)(TargetLocation.Z * 10);
    Ar << X << Y << Z;
    
    bOutSuccess = true;
    return true;
}
```

每命令 ~50-200 字节(根据 Targets 数量)。

## 23.2.2 FMassCommandFragment(entity 内的命令队列)

```cpp
USTRUCT()
struct FMassCommandQueueFragment : public FMassFragment
{
    GENERATED_BODY()
    
    // 等待执行的命令(按顺序)
    TArray<FRTSCommand, TFixedAllocator<8>> PendingCommands;     // 最多 8 命令
    
    // 当前执行的命令
    int32 CurrentIndex = 0;
};
```

`TFixedAllocator<8>` 限制队列长度,避免内存爆炸(8 命令足够 RTS 玩家用)。

## 23.2.3 命令处理器

```cpp
class URTSCommandProcessor : public UMassProcessor
{
    void Execute(...) override
    {
        Query.ForEachEntityChunk(...) [&](FMassExecutionContext& Ctx)
        {
            auto Queues = Ctx.GetMutableFragmentView<FMassCommandQueueFragment>();
            auto Goals = Ctx.GetMutableFragmentView<FMassMovementGoalFragment>();
            auto Targets = Ctx.GetMutableFragmentView<FAttackTargetFragment>();
            
            for entity:
                FMassCommandQueueFragment& Q = Queues[i];
                if (Q.CurrentIndex < Q.PendingCommands.Num())
                {
                    const FRTSCommand& Cmd = Q.PendingCommands[Q.CurrentIndex];
                    
                    switch (Cmd.Type)
                    {
                        case ECmdType::Move:
                            Goals[i].Target = Cmd.TargetLocation;
                            Goals[i].State = EGoal::Moving;
                            break;
                        case ECmdType::Attack:
                            Targets[i].Entity = Cmd.TargetEntity;
                            break;
                        // ...
                    }
                    
                    if (IsCommandComplete(...)) Q.CurrentIndex++;
                }
        });
    }
};
```
