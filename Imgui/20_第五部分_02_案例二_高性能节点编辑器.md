## 5.2.1 数据模型

```cpp
// HazelNodeEditor.h
namespace Hazel {

struct PinId   { int64_t Value; };
struct NodeId  { int64_t Value; };
struct LinkId  { int64_t Value; };

enum class PinKind { Input, Output };
enum class PinType { Flow, Bool, Int, Float, String, Object };

struct Pin {
    PinId   Id;
    PinKind Kind;
    PinType Type;
    std::string Name;
    NodeId  ParentNode;
    
    // 渲染时填充
    ImVec2 Pos;             // 屏幕坐标
};

struct Node {
    NodeId  Id;
    std::string Title;
    ImVec2  Position;       // 画布坐标（不受 zoom/pan 影响）
    ImVec2  Size;           // 屏幕坐标尺寸（zoom 后）
    bool    Selected;
    
    std::vector<Pin> InputPins;
    std::vector<Pin> OutputPins;
};

struct Link {
    LinkId Id;
    PinId  StartPin;       // 输出 pin
    PinId  EndPin;         // 输入 pin
};

struct Editor {
    std::vector<Node> Nodes;
    std::vector<Link> Links;
    
    // 视口状态
    ImVec2 CanvasOrigin;   // 画布原点的屏幕位置
    float  Zoom = 1.0f;    // 缩放系数
    
    // 交互状态
    NodeId DraggingNode = {0};
    PinId  DraggingPin  = {0};
    bool   PanningCanvas = false;
    ImVec2 SelectionStart;
    ImVec2 SelectionEnd;
    bool   IsBoxSelecting = false;
};

}  // namespace Hazel
```

## 5.2.2 Canvas：pan / zoom

### 屏幕坐标 ↔ 画布坐标

```cpp
namespace Hazel {

struct Editor {
    // ...
    
    ImVec2 CanvasToScreen(ImVec2 canvas_pos) const {
        return ImVec2(CanvasOrigin.x + canvas_pos.x * Zoom,
                      CanvasOrigin.y + canvas_pos.y * Zoom);
    }
    
    ImVec2 ScreenToCanvas(ImVec2 screen_pos) const {
        return ImVec2((screen_pos.x - CanvasOrigin.x) / Zoom,
                      (screen_pos.y - CanvasOrigin.y) / Zoom);
    }
};

}
```

### Canvas 渲染入口

```cpp
void Editor::Render(const char* str_id) {
    ImGui::PushID(str_id);
    
    ImGui::BeginChild("Canvas", ImVec2(0, 0),
                      ImGuiChildFlags_None,
                      ImGuiWindowFlags_NoScrollbar | ImGuiWindowFlags_NoScrollWithMouse);
    
    ImVec2 canvas_min = ImGui::GetCursorScreenPos();
    ImVec2 canvas_max = canvas_min + ImGui::GetContentRegionAvail();
    
    // 第一帧初始化：画布原点位于窗口中心
    if (CanvasOrigin.x == 0 && CanvasOrigin.y == 0)
        CanvasOrigin = (canvas_min + canvas_max) * 0.5f;
    
    // 处理输入（pan / zoom）
    HandleCanvasInput(canvas_min, canvas_max);
    
    // 准备 splitter——5 个 channel
    ImDrawList* dl = ImGui::GetWindowDrawList();
    static thread_local ImDrawListSplitter splitter;
    splitter.Split(dl, 5);
    
    // ① Channel 0: 背景网格
    splitter.SetCurrentChannel(dl, 0);
    DrawGrid(canvas_min, canvas_max);
    
    // ② Channel 1: 节点之间的连线
    splitter.SetCurrentChannel(dl, 1);
    DrawLinks();
    
    // ③ Channel 2: 节点框 + 内部
    splitter.SetCurrentChannel(dl, 2);
    DrawNodes();
    
    // ④ Channel 3: 拖动中的临时连线
    splitter.SetCurrentChannel(dl, 3);
    if (DraggingPin.Value != 0)
        DrawDragLink();
    
    // ⑤ Channel 4: 选中框 / 框选矩形
    splitter.SetCurrentChannel(dl, 4);
    if (IsBoxSelecting)
        DrawSelectionRect();
    
    splitter.Merge(dl);
    
    ImGui::EndChild();
    ImGui::PopID();
}
```

**5 个 channel** 让"提交顺序"和"绘制顺序"解耦——你按"节点 → 节点 → 节点 → 节点的连线"的逻辑顺序写代码，但渲染时背景始终在最底、选中框始终在最上。

### 处理 pan + zoom

```cpp
void Editor::HandleCanvasInput(ImVec2 canvas_min, ImVec2 canvas_max) {
    ImGuiIO& io = ImGui::GetIO();
    
    // 鼠标在画布内才响应
    bool hovered = ImGui::IsWindowHovered() && 
                   io.MousePos.x >= canvas_min.x && io.MousePos.x <= canvas_max.x &&
                   io.MousePos.y >= canvas_min.y && io.MousePos.y <= canvas_max.y;
    
    // ① Pan: 中键拖动 / 空格 + 左键
    if (hovered && (ImGui::IsMouseDragging(ImGuiMouseButton_Middle, 0.0f) ||
                    (ImGui::IsKeyDown(ImGuiKey_Space) && ImGui::IsMouseDragging(ImGuiMouseButton_Left, 0.0f)))) {
        ImVec2 delta = io.MouseDelta;
        CanvasOrigin += delta;
        PanningCanvas = true;
    } else {
        PanningCanvas = false;
    }
    
    // ② Zoom: 滚轮
    if (hovered && io.MouseWheel != 0.0f) {
        // 缩放中心 = 鼠标位置（让鼠标下的内容保持在原位）
        ImVec2 mouse_canvas = ScreenToCanvas(io.MousePos);
        
        float old_zoom = Zoom;
        Zoom *= ImPow(1.1f, io.MouseWheel);
        Zoom = ImClamp(Zoom, 0.1f, 5.0f);
        
        // 调整 origin 让 mouse_canvas 仍在 mouse_pos 屏幕位置
        ImVec2 new_screen = CanvasToScreen(mouse_canvas);
        CanvasOrigin += io.MousePos - new_screen;
    }
}
```

**核心 Zoom 算法**：

1. 记录鼠标当前画布坐标 `mouse_canvas`。
2. 改 Zoom。
3. 算出新 Zoom 下 `mouse_canvas` 对应的新屏幕位置 `new_screen`。
4. 调整 `CanvasOrigin` 让 `new_screen == io.MousePos`——即鼠标下的内容**视觉上不动**。

这是所有专业画布工具（Photoshop / Figma / Krita）的标准 zoom 行为。

### 网格绘制

```cpp
void Editor::DrawGrid(ImVec2 canvas_min, ImVec2 canvas_max) {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    
    const float GRID_STEP = 64.0f * Zoom;
    if (GRID_STEP < 4.0f) return;   // zoom 太小时不画
    
    ImU32 col_minor = IM_COL32(80, 80, 80, 80);
    ImU32 col_major = IM_COL32(80, 80, 80, 160);
    
    // 网格起点（找到第一条可见的网格线）
    float x_start = ImFmod(CanvasOrigin.x, GRID_STEP);
    float y_start = ImFmod(CanvasOrigin.y, GRID_STEP);
    
    for (float x = canvas_min.x + x_start; x < canvas_max.x; x += GRID_STEP) {
        bool is_major = ImFmod((x - CanvasOrigin.x) / GRID_STEP, 5.0f) < 0.5f;
        dl->AddLine(ImVec2(x, canvas_min.y), ImVec2(x, canvas_max.y),
                    is_major ? col_major : col_minor);
    }
    for (float y = canvas_min.y + y_start; y < canvas_max.y; y += GRID_STEP) {
        bool is_major = ImFmod((y - CanvasOrigin.y) / GRID_STEP, 5.0f) < 0.5f;
        dl->AddLine(ImVec2(canvas_min.x, y), ImVec2(canvas_max.x, y),
                    is_major ? col_major : col_minor);
    }
}
```

——每 64 像素一条线，每 5 条加粗。zoom 太小时不画（避免线段密度爆炸）。

## 5.2.3 节点：DrawNodes

```cpp
void Editor::DrawNodes() {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    
    for (Node& node : Nodes) {
        ImGui::PushID((int)node.Id.Value);
        DrawSingleNode(node);
        ImGui::PopID();
    }
}

void Editor::DrawSingleNode(Node& node) {
    ImGui::PushID("Node");
    
    // 1. 转换坐标
    ImVec2 screen_pos = CanvasToScreen(node.Position);
    
    // 2. 算节点尺寸（基于内容，简化处理）
    float title_height = ImGui::GetFontSize() + 8.0f;
    float pin_row_h = ImGui::GetFontSize() + 4.0f;
    int max_pins = ImMax((int)node.InputPins.size(), (int)node.OutputPins.size());
    float content_h = max_pins * pin_row_h;
    ImVec2 node_size = ImVec2(180.0f * Zoom, (title_height + content_h) * Zoom);
    node.Size = node_size;
    
    // 3. 三段式：ID + bb + ItemAdd + ButtonBehavior
    ImGuiID id = ImGui::GetID("Body");
    ImRect bb(screen_pos, screen_pos + node_size);
    ImGui::ItemSize(ImVec2(0, 0));   // 不占用 layout 空间（自由定位）
    ImGui::SetCursorScreenPos(screen_pos);   // 但保持 LastItemData 正确
    ImGui::ItemAdd(bb, id);
    
    bool hovered, held;
    bool clicked = ImGui::ButtonBehavior(bb, id, &hovered, &held,
                                         ImGuiButtonFlags_PressedOnClick |
                                         ImGuiButtonFlags_AllowOverlap);
    
    // 4. 拖动检测
    if (held && ImGui::IsMouseDragging(ImGuiMouseButton_Left, 0.0f)) {
        ImVec2 delta = ImGui::GetIO().MouseDelta;
        // delta 是屏幕坐标——除以 zoom 转成画布坐标
        node.Position += delta / Zoom;
    }
    
    // 5. 单击 → 选中
    if (clicked) {
        if (!ImGui::GetIO().KeyCtrl) {
            // 清其他选中
            for (Node& n : Nodes) n.Selected = false;
        }
        node.Selected = !node.Selected;
    }
    
    // 6. 渲染
    ImDrawList* dl = ImGui::GetWindowDrawList();
    
    // 节点背景
    ImU32 bg_col = node.Selected ? IM_COL32(80, 100, 140, 220) : IM_COL32(50, 50, 60, 220);
    if (hovered) bg_col = IM_COL32(70, 70, 80, 220);
    dl->AddRectFilled(bb.Min, bb.Max, bg_col, 4.0f * Zoom);
    
    // 节点边框
    ImU32 border_col = node.Selected ? IM_COL32(255, 200, 100, 255) : IM_COL32(20, 20, 30, 255);
    dl->AddRect(bb.Min, bb.Max, border_col, 4.0f * Zoom, 0, 2.0f);
    
    // 标题栏
    ImVec2 title_bb_max = ImVec2(bb.Max.x, bb.Min.y + title_height * Zoom);
    dl->AddRectFilled(bb.Min, title_bb_max, IM_COL32(30, 30, 40, 255), 4.0f * Zoom);
    
    // 标题文字
    dl->AddText(bb.Min + ImVec2(8, 4) * Zoom, IM_COL32_WHITE, node.Title.c_str());
    
    // 端口
    for (size_t i = 0; i < node.InputPins.size(); i++) {
        ImVec2 pin_pos = bb.Min + ImVec2(0, (title_height + i * pin_row_h + pin_row_h * 0.5f) * Zoom);
        node.InputPins[i].Pos = pin_pos;
        DrawPin(node.InputPins[i]);
    }
    for (size_t i = 0; i < node.OutputPins.size(); i++) {
        ImVec2 pin_pos = bb.Max - ImVec2(0, node.Size.y - (title_height + i * pin_row_h + pin_row_h * 0.5f) * Zoom);
        pin_pos.x = bb.Max.x;   // 右侧
        node.OutputPins[i].Pos = pin_pos;
        DrawPin(node.OutputPins[i]);
    }
    
    ImGui::PopID();
}
```

**关键技巧**：

1. `ItemSize(ImVec2(0, 0))` + `SetCursorScreenPos(screen_pos)` —— 让节点**不占用 ImGui Layout 空间**（节点位置自管），但保持 ItemAdd 的语义（让 Hover/Click 工作）。
2. `ImGuiButtonFlags_AllowOverlap` —— 让节点和端口可以重叠（端口提交在节点之后，但仍能 hover）。
3. 拖动 delta 从屏幕坐标除以 Zoom 转回画布坐标——画布坐标是"逻辑坐标"，与 zoom 无关。

## 5.2.4 端口（Pin）：DrawPin + 连接交互

```cpp
void Editor::DrawPin(Pin& pin) {
    ImGui::PushID((int)pin.Id.Value);
    
    float radius = 6.0f * Zoom;
    ImRect bb(pin.Pos - ImVec2(radius, radius), pin.Pos + ImVec2(radius, radius));
    
    ImGuiID id = ImGui::GetID("##Pin");
    ImGui::ItemAdd(bb, id);
    
    bool hovered, held;
    bool clicked = ImGui::ButtonBehavior(bb, id, &hovered, &held,
                                         ImGuiButtonFlags_PressedOnClick);
    
    // 鼠标按下 pin → 开始拖出连线
    if (clicked) {
        DraggingPin = pin.Id;
    }
    
    // 鼠标抬起在另一个 pin 上 → 完成连接
    if (DraggingPin.Value != 0 && !ImGui::IsMouseDown(ImGuiMouseButton_Left)) {
        if (hovered && pin.Id.Value != DraggingPin.Value) {
            CreateLink(DraggingPin, pin.Id);
        }
        if (!hovered)   // 没命中——保持拖动状态等待用户继续
            ;
    }
    
    // 渲染
    ImDrawList* dl = ImGui::GetWindowDrawList();
    ImU32 col = GetPinColor(pin.Type);
    if (hovered) col = IM_COL32(255, 255, 100, 255);
    
    bool is_connected = IsConnected(pin.Id);
    if (is_connected) {
        dl->AddCircleFilled(pin.Pos, radius, col);
    } else {
        dl->AddCircle(pin.Pos, radius, col, 0, 2.0f);
    }
    
    ImGui::PopID();
}
```

## 5.2.5 连线：DrawLinks

```cpp
void Editor::DrawLinks() {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    
    for (const Link& link : Links) {
        Pin* p_start = FindPin(link.StartPin);
        Pin* p_end   = FindPin(link.EndPin);
        if (!p_start || !p_end) continue;
        
        DrawConnection(p_start->Pos, p_end->Pos, GetPinColor(p_start->Type));
    }
}

void Editor::DrawConnection(ImVec2 from, ImVec2 to, ImU32 col, float thickness) {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    
    // Bezier 控制点（水平方向偏移半个距离）
    float dx = ImAbs(to.x - from.x) * 0.5f;
    ImVec2 cp1 = from + ImVec2(dx, 0);
    ImVec2 cp2 = to   - ImVec2(dx, 0);
    
    dl->AddBezierCubic(from, cp1, cp2, to, col, thickness * Zoom, 0);
}
```

**为什么 Bezier**：直线连接看起来生硬，曲线更符合"信号流"的视觉直觉。控制点放在水平中点处，给出标准的"S 形"曲线。

```cpp
void Editor::HandleSelectionInput(ImVec2 canvas_min, ImVec2 canvas_max) {
    ImGuiIO& io = ImGui::GetIO();
    
    // 在画布空白区域按下左键 + 没有按 Space → 开始框选
    bool space_held = ImGui::IsKeyDown(ImGuiKey_Space);
    if (!space_held && ImGui::IsMouseClicked(ImGuiMouseButton_Left) &&
        ImGui::IsWindowHovered() && /*没点到节点 / pin*/) {
        IsBoxSelecting = true;
        SelectionStart = io.MousePos;
        SelectionEnd = io.MousePos;
        if (!io.KeyCtrl) {
            for (Node& n : Nodes) n.Selected = false;
        }
    }
    
    if (IsBoxSelecting) {
        SelectionEnd = io.MousePos;
        ImVec2 sel_min = ImMin(SelectionStart, SelectionEnd);
        ImVec2 sel_max = ImMax(SelectionStart, SelectionEnd);
        
        // 检查每个节点是否与选框相交
        for (Node& n : Nodes) {
            ImVec2 n_pos_screen = CanvasToScreen(n.Position);
            ImRect n_bb(n_pos_screen, n_pos_screen + n.Size);
            ImRect sel_bb(sel_min, sel_max);
            n.Selected = n_bb.Overlaps(sel_bb);
        }
        
        if (!ImGui::IsMouseDown(ImGuiMouseButton_Left)) {
            IsBoxSelecting = false;
        }
    }
}

void Editor::DrawSelectionRect() {
    ImDrawList* dl = ImGui::GetWindowDrawList();
    ImVec2 sel_min = ImMin(SelectionStart, SelectionEnd);
    ImVec2 sel_max = ImMax(SelectionStart, SelectionEnd);
    
    dl->AddRectFilled(sel_min, sel_max, IM_COL32(100, 150, 255, 50));
    dl->AddRect(sel_min, sel_max, IM_COL32(100, 150, 255, 200), 0, 0, 2.0f);
}
```

## 5.2.7 ID 命名空间设计

回到第 4.1 章——ID 冲突是节点编辑器最容易出问题的地方。**1000 节点 × 多个 pin** 的命名空间必须严格设计。

### 推荐设计

```cpp
void Editor::Render(const char* str_id) {
    ImGui::PushID(str_id);              // 整个编辑器命名空间
    // ...
    
    // 每个节点：
    for (Node& node : Nodes) {
        ImGui::PushID((int)node.Id.Value);   // 节点级命名空间
        DrawSingleNode(node);
        ImGui::PopID();
    }
    
    ImGui::PopID();
}

void Editor::DrawSingleNode(Node& node) {
    // 节点 body
    ImGuiID body_id = ImGui::GetID("Body");
    
    // 端口（每个 pin 自己的 PushID）
    for (Pin& pin : node.InputPins) {
        ImGui::PushID((int)pin.Id.Value);   // pin 级命名空间
        ImGuiID pin_id = ImGui::GetID("##Pin");
        // ...
        ImGui::PopID();
    }
}
```

**最终 ID 路径**：

```
Editor("MyEditor") → Node(42) → Pin(1024) → "##Pin"
   ↓                    ↓             ↓        ↓
 hash             hash         hash     hash
```

——4 层嵌套，每层独立，永远不冲突。

### Pin ID 用 `int` vs 用 `pin_id_64` 哈希

`PushID(int)` 哈希 `int` 的 4 字节。如果你的 PinId 是 64-bit：

```cpp
PinId  pin_id = ...;  // int64_t
ImGuiID imgui_id = ImHashData(&pin_id, sizeof(pin_id), seed);
ImGui::PushOverrideID(imgui_id);
```

——**推荐**这种做法，避免 64→32 截断的潜在碰撞。

## 5.2.8 性能：1000 节点 60fps

### 顶点数估算

每个节点（带 6 个 pin）：

- 节点框：4 顶点 × 2（背景 + 边框）= 8 顶点
- 标题栏背景：4
- 标题文字：5 字符 × 4 = 20
- 6 个 pin × 圆形（16 段）= 6 × 32 = 192
- 平均每节点 ~225 顶点

每条连线（Bezier）：
- 默认 tess_tol 下约 30~50 顶点

1000 节点 + 1500 连线 ≈ **300K 顶点 / 帧**。

GPU 上传 + 绘制 30 万顶点 / 帧 = **~ 1 ms**（桌面 GPU）。这不是瓶颈。

### CPU 瓶颈：ItemAdd / ButtonBehavior 调用次数

1000 节点 × 7 次 ItemAdd（节点 + 6 pin）= **7000 次 / 帧**。每次 ItemAdd ~ 0.5μs → **3.5 ms / 帧**。

这就是瓶颈。优化方向：

### 优化 1：粗粒度剔除

**画布外**的节点完全不调 widget API：

```cpp
void Editor::DrawNodes() {
    ImVec2 canvas_min = ImGui::GetWindowPos();
    ImVec2 canvas_max = canvas_min + ImGui::GetWindowSize();
    
    for (Node& node : Nodes) {
        ImVec2 node_screen_min = CanvasToScreen(node.Position);
        ImVec2 node_screen_max = node_screen_min + node.Size;
        
        // 完全在画布外 —— 跳过
        if (node_screen_max.x < canvas_min.x || node_screen_min.x > canvas_max.x ||
            node_screen_max.y < canvas_min.y || node_screen_min.y > canvas_max.y)
            continue;
        
        ImGui::PushID((int)node.Id.Value);
        DrawSingleNode(node);
        ImGui::PopID();
    }
}
```

——只画可见节点。1000 节点中通常只有几十在画布内 → ItemAdd 调用降到几百。

### 优化 2：连线 LOD

zoom < 0.3 时：

```cpp
if (Zoom < 0.3f) {
    // 用直线代替 bezier
    dl->AddLine(from, to, col, thickness);
} else {
    dl->AddBezierCubic(from, cp1, cp2, to, col, thickness, num_segments);
}
```

——zoom 越小，曲线越细分越浪费。直线在远距离看不出差别。

### 优化 3：连线 batch 合并

如果你大量连线**同色 + 同粗细**，可以合并到一个 batch：

```cpp
// 把所有同色连线放在同一个 channel 内
splitter.SetCurrentChannel(dl, 1);
for (const Link& link : Links) {
    DrawConnection(...);   // 全部在 channel 1
}
```

——通过 splitter 的合并能力，相同 ClipRect / TexRef 的连线合并成 1 cmd。

### 优化 4：避免每帧重新计算

```cpp
struct Node {
    // ...
    int LastFrameUpdated;
    ImRect CachedScreenBB;   // 缓存屏幕 bbox
};

// 节点位置 / Zoom / Pan 没变 → 用缓存
if (node.LastFrameUpdated == g.FrameCount - 1 && /*没变*/) {
    return node.CachedScreenBB;
}
```

——但需要谨慎处理"何时失效"。除非真的成为瓶颈，建议简单实现先。

### 真实数据

`imgui-node-editor` 的优化版本能撑 5000 节点 60fps。`imnodes` 简单版本 1000 节点 30fps。本章给的实现合理优化后可达 **2000 节点 60fps**。

### 复制粘贴

```cpp
void Editor::CopySelected() {
    ClipboardData clip;
    for (Node& n : Nodes)
        if (n.Selected)
            clip.Nodes.push_back(SerializeNode(n));
    
    // 同时复制相关 link
    for (const Link& link : Links) {
        if (IsSelectedPin(link.StartPin) && IsSelectedPin(link.EndPin))
            clip.Links.push_back(SerializeLink(link));
    }
    
    SaveToClipboard(clip);
}

void Editor::Paste() {
    ClipboardData clip = LoadFromClipboard();
    
    // 偏移粘贴的节点（避免直接覆盖原节点）
    ImVec2 offset(20, 20);
    
    // 重新分配 ID + 修复 link 引用
    std::map<NodeId, NodeId> id_map;
    std::map<PinId, PinId> pin_id_map;
    
    for (NodeData& nd : clip.Nodes) {
        Node new_node = DeserializeNode(nd);
        NodeId new_id = AllocateNodeId();
        id_map[new_node.Id] = new_id;
        new_node.Id = new_id;
        new_node.Position += offset;
        new_node.Selected = true;   // 选中粘贴的节点
        
        // pin id 也重新分配
        for (Pin& p : new_node.InputPins) {
            PinId new_pin = AllocatePinId();
            pin_id_map[p.Id] = new_pin;
            p.Id = new_pin;
            p.ParentNode = new_id;
        }
        // ... output pins 类似
        
        Nodes.push_back(new_node);
    }
    
    // 重连接 link
    for (LinkData& ld : clip.Links) {
        Link new_link;
        new_link.Id = AllocateLinkId();
        new_link.StartPin = pin_id_map[ld.StartPin];
        new_link.EndPin = pin_id_map[ld.EndPin];
        Links.push_back(new_link);
    }
}
```

## 5.2.10 撤销 / 重做

ImGui 自身不提供 Undo——节点编辑器需要自己实现。最简单方案：**命令模式**。

```cpp
struct Command {
    virtual ~Command() {}
    virtual void Execute() = 0;
    virtual void Undo() = 0;
};

struct AddNodeCommand : Command {
    Editor* editor;
    Node    node;   // 含完整数据
    
    void Execute() override { editor->Nodes.push_back(node); }
    void Undo() override {
        editor->Nodes.erase(/* find by id */);
    }
};

struct MoveNodeCommand : Command {
    Editor* editor;
    NodeId  id;
    ImVec2  old_pos;
    ImVec2  new_pos;
    
    void Execute() override { editor->FindNode(id)->Position = new_pos; }
    void Undo() override    { editor->FindNode(id)->Position = old_pos; }
};

class UndoStack {
    std::vector<std::unique_ptr<Command>> history;
    int current = -1;
    
public:
    void Push(std::unique_ptr<Command> cmd) {
        history.resize(current + 1);
        cmd->Execute();
        history.push_back(std::move(cmd));
        current++;
    }
    
    void Undo() {
        if (current >= 0) {
            history[current]->Undo();
            current--;
        }
    }
    
    void Redo() {
        if (current + 1 < (int)history.size()) {
            current++;
            history[current]->Execute();
        }
    }
};
```

每次"用户操作"提交一个 Command 而不是直接改数据。Ctrl+Z / Ctrl+Y 触发 Undo / Redo。

```cpp
namespace Hazel {

void NodeEditor::Render(const char* str_id) {
    ImGui::PushID(str_id);
    
    ImGui::BeginChild("Canvas", ImVec2(0, 0),
                      ImGuiChildFlags_None,
                      ImGuiWindowFlags_NoScrollbar | ImGuiWindowFlags_NoScrollWithMouse);
    
    ImVec2 canvas_min = ImGui::GetCursorScreenPos();
    ImVec2 canvas_max = canvas_min + ImGui::GetContentRegionAvail();
    if (m_CanvasOrigin.x == 0) m_CanvasOrigin = (canvas_min + canvas_max) * 0.5f;
    
    HandleCanvasInput(canvas_min, canvas_max);
    HandleSelectionInput(canvas_min, canvas_max);
    HandleHotkeys();    // Delete / Ctrl+C / Ctrl+V / Ctrl+Z / Ctrl+Y
    
    ImDrawList* dl = ImGui::GetWindowDrawList();
    dl->PushClipRect(canvas_min, canvas_max);   // 限制绘制在画布内
    
    // 5 个 channel
    static thread_local ImDrawListSplitter splitter;
    splitter.Split(dl, 5);
    
    splitter.SetCurrentChannel(dl, 0); DrawGrid(canvas_min, canvas_max);
    splitter.SetCurrentChannel(dl, 1); DrawLinks();
    splitter.SetCurrentChannel(dl, 2); DrawNodes();
    splitter.SetCurrentChannel(dl, 3); if (m_DraggingPin.Value) DrawDragLink();
    splitter.SetCurrentChannel(dl, 4); if (m_IsBoxSelecting) DrawSelectionRect();
    
    splitter.Merge(dl);
    
    dl->PopClipRect();
    
    ImGui::EndChild();
    ImGui::PopID();
}

}  // namespace Hazel
```

## 5.2.14 本章小结

节点编辑器的 6 个核心子系统：

1. **Canvas (pan/zoom)**：屏幕 ↔ 画布坐标转换 + 鼠标输入处理。
2. **Nodes**：用三段式 widget 模板 + ImGuiButtonFlags_AllowOverlap。
3. **Pins**：嵌入节点内部，用 ID 命名空间隔离。
4. **Links**：Bezier 曲线 + 拖动中临时线。
5. **Selection**：框选 + 单选 / 多选状态。
6. **5-channel Splitter**：背景/连线/节点/拖线/选框分通道，最终合并成最少 cmd。

性能优化关键：

- 画布外节点完全跳过（`ItemAdd` 调用是 CPU 瓶颈）。
- zoom 太小时用直线代替 bezier。
- 同色连线 batch 合并。
- 节点 ID 用全 64-bit 哈希避免碰撞。
